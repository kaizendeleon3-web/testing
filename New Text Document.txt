    console.log('Firebase offline — using local storage:', e);
    firebaseAvailable = false;
  }

  // ===== RENDER PRODUCTS WITH PHOTO =====
  function renderCat(catId, items){
    const g = document.getElementById(catId + '-grid');
    if(!g) return;
    g.innerHTML = items.map(p => {
      // 📸 PHOTO DISPLAY: Shows your .jpg or placeholder
      const photoHtml = p.photo 
        ? `<img src="${p.photo}" alt="${p.name}" onerror="this.parentElement.innerHTML='<div class=prod-img-placeholder>${p.label}</div>'">` 
        : `<div class="prod-img-placeholder">${p.label}</div>`;

      return '<div class="product-card">' +
        '<div class="prod-img">' + photoHtml + '</div>' +
        '<div class="prod-body">' +
          '<h3 class="prod-name">' + p.name + '</h3>' +
          '<div class="prod-price">P' + p.price.toLocaleString() + '</div>' +
          '<div class="size-buttons">' +
            p.sizes.split(',').map(s => '<button class="size-btn" data-prod="' + p.id + '" data-size="' + s.trim() + '" onclick="pickSize(\'' + p.id + '\',\'' + s.trim() + '\',this)">' + s.trim() + '</button>').join('') +
          '</div>' +
          '<button class="btn btn-chrome btn-block" onclick="addItem(\'' + p.id + '\',\'' + p.name.replace(/'/g,"\\'") + '\',' + p.price + ')">ADD TO CART</button>' +
        '</div>' +
      '</div>';
    }).join('');
  }

  function renderAllProducts(){
    renderCat('decks', PRODUCTS.decks);
    renderCat('trucks', PRODUCTS.trucks);
    renderCat('wheels', PRODUCTS.wheels);
    renderCat('grip', PRODUCTS.grip);
    renderCat('bearings', PRODUCTS.bearings);
    renderCat('tees', PRODUCTS.tees);
  }

  // ===== CART FUNCTIONS =====
  window.pickSize = function(id, sz, btn){
    document.querySelectorAll('.size-btn[data-prod="' + id + '"]').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
    selectedSize[id] = sz;
  }

  window.addItem = function(id, name, price){
    const sz = selectedSize[id] || 'Standard';
    const idx = cart.findIndex(x => x.id === id && x.size === sz);
    idx >= 0 ? cart[idx].qty++ : cart.push({id, name, price, size: sz, qty: 1});
    saveCart();
    renderCart();
  }

  function saveCart(){
    localStorage.setItem('_paradise_cart', JSON.stringify(cart));
  }

  function renderCart(){
    const qtyTotal = cart.reduce((s, i) => s + i.qty, 0);
    const countEl = document.getElementById('cart-count');
    if(countEl) countEl.innerText = qtyTotal;
    const cartItemsEl = document.getElementById('cart-items');
    if(!cartItemsEl) return;
    if(!cart.length){
      cartItemsEl.innerHTML = '<p style="color:#666">Cart is empty — pick a size then click ADD TO CART</p>';
      updateTotals(0, SHIP_STANDARD);
      return;
    }
    cartItemsEl.innerHTML = cart.map((item, i) =>
      '<div class="cart-item"><div><strong>' + item.name + '</strong><br>Size: ' + item.size + ' — P' + item.price.toLocaleString() + ' ea.</div><div class="cart-qty"><button class="qty-btn" onclick="chgQty(' + i + ',-1)">−</button>' + item.qty + '<button class="qty-btn" onclick="chgQty(' + i + ',1)">+</button></div></div>'
    ).join('');
    const sub = cart.reduce((s, i) => s + i.price * i.qty, 0);
    const ship = sub >= FREE_SHIP_AT ? 0 : SHIP_STANDARD;
    updateTotals(sub, ship);
  }

  window.chgQty = function(i, d){
    cart[i].qty += d;
    if(cart[i].qty < 1) cart.splice(i, 1);
    saveCart();
    renderCart();
  }

  function updateTotals(sub, ship){
    document.getElementById('subtotal').innerText = 'P' + sub.toLocaleString();
    document.getElementById('shipping').innerText = ship ? 'P' + ship : 'FREE';
    document.getElementById('total').innerText = 'P' + (sub + ship).toLocaleString();
  }

  // ===== CHECKOUT & PLACE ORDER =====
  window.openCheckout = function(){
    if(!cart.length){ alert('Add items to cart first!'); return; }
    document.getElementById('checkout-modal').classList.add('show');
  }
  window.closeCheckout = function(){
    document.getElementById('checkout-modal').classList.remove('show');
  }

  window.placeOrder = function(){
    const n = document.getElementById('del-name').value.trim();
    const ph = document.getElementById('del-phone').value.trim();
    const a = document.getElementById('del-address').value.trim();
    const opt = document.getElementById('del-option').value;
    const r = document.getElementById('ref-number').value.trim();
    const errEl = document.getElementById('checkout-err');
    if(!n){ errEl.innerText='Enter your name!'; return; }
    if(!ph){ errEl.innerText='Enter your phone!'; return; }
    if(!a){ errEl.innerText='Enter your address!'; return; }
    if(!r){ errEl.innerText='Enter GCash Reference #!'; return; }

    const sub = cart.reduce((s, i) => s + i.price * i.qty, 0);
    let shipAmt = opt === 'pickup' ? 0 : (opt === 'express' ? SHIP_EXPRESS : SHIP_STANDARD);
    if(sub >= FREE_SHIP_AT) shipAmt = 0;

    const ts = Date.now();
    const newOrder = {
      id: ts.toString(36).toUpperCase(),
      localKey: 'local_' + ts,
      name: n, phone: ph, address: a, option: opt,
      shipping: shipAmt, refNum: r,
      items: JSON.parse(JSON.stringify(cart)),
      subtotal: sub, total: sub + shipAmt,
      date: new Date().toLocaleString(),
      timestamp: ts, status: 'pending'
    };

    const onSuccess = function(){
      alert('✅ ORDER SENT TO ADMIN INBOX!\n\nOrder #: ' + newOrder.id);
      cart = []; saveCart(); renderCart(); closeCheckout();
      document.getElementById('del-name').value='';
      document.getElementById('del-phone').value='';
      document.getElementById('del-address').value='';
      document.getElementById('ref-number').value='';
    };

    // Save to Firebase OR local storage
    if(firebaseAvailable && database){
      database.ref('orders').push(newOrder).then(onSuccess).catch(err => {
        console.log('Firebase error, saved locally:', err);
        localOrders.push(newOrder);
        localStorage.setItem(LOCAL_ORDERS_KEY, JSON.stringify(localOrders));
        onSuccess();
      });
    } else {
      localOrders.push(newOrder);
      localStorage.setItem(LOCAL_ORDERS_KEY, JSON.stringify(localOrders));
      onSuccess();
    }
  }

  // ===== ADMIN LOGIN & ORDERS =====
  function startSession(){ document.body.classList.add('admin-mode'); localStorage.setItem(SESSION_KEY, JSON.stringify({active:true})); }

    if(s && s.active) startSession();
  }
  window.adminLogout = function(){ localStorage.removeItem(SESSION_KEY); document.body.classList.remove('admin-mode'); alert('Logged out!'); }
  window.openAuthModal = function(){ document.getElementById('auth-modal').classList.add('show'); }
  window.closeAuthModal = function(){ document.getElementById('auth-modal').classList.remove('show'); }

    const p = document.getElementById('admin-pass-input').value;
    if(p === DEFAULT_ADMIN_PASS){
      startSession(); closeAuthModal(); alert('✅ ADMIN MODE UNLOCKED!\n\nClick ORDERS to view inbox.');

  }

  function updateOrderCount(){
    if(firebaseAvailable && database){
      database.ref('orders').orderByChild('status').equalTo('pending').on('value', s => {
        document.getElementById('order-count').innerText = s.numChildren();
      });
    } else {
      document.getElementById('order-count').innerText = localOrders.filter(o=>o.status==='pending').length;
    }
  }

  window.openOrdersInbox = function(){
    if(!document.body.classList.contains('admin-mode')){ alert('⚠️ Login as Admin first!'); return; }
    document.getElementById('orders-modal').classList.add('show');
    renderOrdersList();
  }
  window.closeOrdersInbox = function(){ document.getElementById('orders-modal').classList.remove('show'); }

  function renderOrdersList(){
    const list = document.getElementById('orders-list');
    const display = orders => {
      orders.sort((a,b) => (a.status==='pending'?-1:1) || b.timestamp - a.timestamp);
      if(!orders.length){ list.innerHTML = '<div class="empty-inbox">📭 NO ORDERS YET</div>'; return; }
      list.innerHTML = orders.map(o => {
        const cls = o.status==='paid'?'paid':'pending';
        const status = o.status==='paid'?'✅ FULLY PAID':'⏳ PENDING';
        const items = o.items.map(i=>'• '+i.name+' — '+i.size+' × '+i.qty+' = P'+(i.price*i.qty).toLocaleString()).join('<br>');
        const key = o.firebaseKey || o.localKey;
        return '<div class="order-card '+cls+'">' +
          '<div class="order-header"><span class="order-id">ORDER #'+o.id+'</span><span class="order-status status-'+cls+'">'+status+'</span></div>' +
          '<div class="order-details">' +
            '<p><strong>Name:</strong> '+o.name+'</p>' +
            '<p><strong>Phone:</strong> '+o.phone+'</p>' +
            '<p><strong>Address:</strong> '+o.address+'</p>' +
            '<p><strong>GCash Ref:</strong> '+o.refNum+'</p>' +
            '<p><strong>TOTAL:</strong> P'+o.total.toLocaleString()+'</p>' +
            '<div class="order-items"><strong>ITEMS:</strong><br><br>'+items+'</div>' +
          '</div>' +
          '<div class="order-actions">' +
            (o.status!=='paid'?'<button class="btn btn-sm btn-paid" onclick="markAsPaid(\''+key+'\')">✅ MARK PAID</button>':'<span style="padding:.45rem .8rem;color:green">✓ PAID</span>') +
            '<button class="btn btn-sm btn-delete" onclick="deleteOrder(\''+key+'\')">🗑️ DELETE</button>' +
          '</div></div>';
      }).join('');
    };

    if(firebaseAvailable && database){
      database.ref('orders').orderByChild('timestamp').once('value', s=>{
        const arr=[]; s.forEach(c=>arr.push({...c.val(),firebaseKey:c.key})); display(arr);
      });
    } else { display([...localOrders]); }
  }

  window.markAsPaid = function(key){
    if(firebaseAvailable && database){
      database.ref('orders/'+key).update({status:'paid'}).then(()=>{renderOrdersList();alert('✅ MARKED AS PAID!');});
    } else {
      const i = localOrders.findIndex(o=>o.localKey===key);
      if(i>=0){ localOrders[i].status='paid'; localStorage.setItem(LOCAL_ORDERS_KEY, JSON.stringify(localOrders)); renderOrdersList(); alert('✅ MARKED AS PAID!'); }
    }
  }

  window.deleteOrder = function(key){
    if(!confirm('Delete this order?')) return;
    if(firebaseAvailable && database){
      database.ref('orders/'+key).remove().then(()=>{renderOrdersList();alert('🗑️ DELETED!');});
    } else {
      localOrders = localOrders.filter(o=>o.localKey!==key);
      localStorage.setItem(LOCAL_ORDERS_KEY, JSON.stringify(localOrders));
      renderOrdersList(); alert('🗑️ DELETED!');
    }
  }

  // ===== INIT =====
  function init(){
    checkSession();
    renderAllProducts();
    renderCart();
    updateOrderCount();
    document.getElementById('loading-overlay').style.display = 'none';
    if(firebaseAvailable && database){
      document.getElementById('sync-status').innerText='ONLINE';
      document.getElementById('sync-status').className='sync-status online';
    } else {
      document.getElementById('sync-status').innerText='LOCAL MODE';
      document.getElementById('sync-status').className='sync-status offline';
    }
  }

  window.onload = init;
})();
